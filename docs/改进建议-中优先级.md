### 1.1 批量操作支持增强

**问题描述**: 当前批量操作功能有限，不支持复杂场景。

**影响文件**: 
- `src/components/AdbTools/AppManagerPanel.tsx`
- `src-tauri/src/commands.rs`

**当前限制**:
- 批量安装应用时，失败一个就停止
- 不支持批量设备操作（同时操作多台设备）
- 没有操作队列管理
- 缺少操作进度详情

**技术方案**:

```typescript
// 前端：批量操作管理器
// src/services/batchOperationService.ts

export interface BatchOperation<T, R> {
  id: string;
  type: 'install' | 'uninstall' | 'transfer' | 'command';
  targets: T[];  // 操作目标（设备或文件）
  operation: (target: T) => Promise<R>;
  strategy: 'sequential' | 'parallel' | 'failfast';
  maxConcurrency?: number;
}

export interface BatchProgress<T, R> {
  total: number;
  completed: number;
  failed: number;
  results: Map<T, { success: boolean; result?: R; error?: string }>;
  currentlyProcessing: T[];
}

export class BatchOperationManager {
  async execute<T, R>(operation: BatchOperation<T, R>): Promise<BatchProgress<T, R>> {
    const progress: BatchProgress<T, R> = {
      total: operation.targets.length,
      completed: 0,
      failed: 0,
      results: new Map(),
      currentlyProcessing: [],
    };

    switch (operation.strategy) {
      case 'sequential':
        return this.executeSequential(operation, progress);
      case 'parallel':
        return this.executeParallel(operation, progress);
      case 'failfast':
        return this.executeFailFast(operation, progress);
    }
  }

  private async executeParallel<T, R>(
    operation: BatchOperation<T, R>,
    progress: BatchProgress<T, R>
  ): Promise<BatchProgress<T, R>> {
    const concurrency = operation.maxConcurrency || 3;
    const chunks = this.chunkArray(operation.targets, concurrency);

    for (const chunk of chunks) {
      progress.currentlyProcessing = chunk;
      const results = await Promise.allSettled(
        chunk.map(target => operation.operation(target))
      );

      results.forEach((result, index) => {
        const target = chunk[index];
        if (result.status === 'fulfilled') {
          progress.results.set(target, { success: true, result: result.value });
          progress.completed++;
        } else {
          progress.results.set(target, { success: false, error: result.reason });
          progress.failed++;
        }
      });

      // 通知进度更新
      this.notifyProgress(progress);
    }

    progress.currentlyProcessing = [];
    return progress;
  }
}

// 使用示例
const batchInstall: BatchOperation<string, void> = {
  id: 'batch-install-' + Date.now(),
  type: 'install',
  targets: apkFiles,
  operation: async (apkPath) => {
    await invoke('install_apk', { apkPath, deviceId: selectedDevice.serial });
  },
  strategy: 'parallel',
  maxConcurrency: 3,
};

const result = await batchOperationManager.execute(batchInstall);
console.log(`成功: ${result.completed}, 失败: ${result.failed}`);
```

#[derive(Debug, Serialize, Deserialize)]
pub struct BatchOperationRequest {
    pub operation_type: String,
    pub targets: Vec<String>,
    pub parameters: serde_json::Value,
    pub strategy: BatchStrategy,
}

#[derive(Debug, Serialize, Deserialize)]
pub enum BatchStrategy {
    Sequential,
    Parallel { max_concurrency: usize },
    FailFast,
}

#[derive(Debug, Serialize, Deserialize)]
pub struct BatchOperationResult {
    pub total: usize,
    pub completed: usize,
    pub failed: usize,
    pub results: HashMap<String, OperationResult>,
}

#[tauri::command]
pub async fn execute_batch_operation(
    request: BatchOperationRequest,
) -> Result<BatchOperationResult> {
    match request.strategy {
        BatchStrategy::Parallel { max_concurrency } => {
            execute_parallel_batch(request, max_concurrency).await
        }
        BatchStrategy::Sequential => {
            execute_sequential_batch(request).await
        }
        BatchStrategy::FailFast => {
            execute_failfast_batch(request).await
        }
    }

### 1.3 操作历史记录和撤销

**问题描述**: 危险操作（删除文件、卸载应用）没有撤销功能。

**技术方案**:

```typescript
// src/services/operationHistoryService.ts

export interface Operation {
  id: string;
  type: 'install' | 'uninstall' | 'delete_file' | 'modify_file';
  timestamp: Date;
  deviceId: string;
  parameters: any;
  result: 'success' | 'failed' | 'cancelled';
  canUndo: boolean;
  undoData?: any;  // 用于撤销的数据
}

export class OperationHistoryService {
  private history: Operation[] = [];
  private maxHistory = 50;

  async recordOperation(operation: Omit<Operation, 'id' | 'timestamp'>): Promise<void> {
    const op: Operation = {
      ...operation,
      id: `op-${Date.now()}`,
      timestamp: new Date(),
    };

    this.history.unshift(op);
    if (this.history.length > this.maxHistory) {
      this.history.pop();
    }

    // 持久化到本地存储
    await this.saveHistory();
  }

  async undoOperation(operationId: string): Promise<void> {
    const operation = this.history.find(op => op.id === operationId);
    if (!operation || !operation.canUndo) {
      throw new Error('无法撤销此操作');
    }

    switch (operation.type) {
      case 'uninstall':
        // 重新安装应用（如果有备份）
        if (operation.undoData?.apkPath) {
          await invoke('install_apk', {
            apkPath: operation.undoData.apkPath,
            deviceId: operation.deviceId,
          });
        }
        break;

      case 'delete_file':
        // 恢复文件（从回收站）
        if (operation.undoData?.backupPath) {
          await invoke('restore_file', {
            backupPath: operation.undoData.backupPath,
            originalPath: operation.parameters.filePath,
            deviceId: operation.deviceId,
          });
        }
        break;

      // 其他操作类型...
    }

    // 标记为已撤销
    operation.result = 'cancelled';
    await this.saveHistory();
  }

  getHistory(deviceId?: string): Operation[] {
    if (deviceId) {
      return this.history.filter(op => op.deviceId === deviceId);
    }
    return this.history;
  }
}

// 在执行危险操作前自动备份
async function uninstallAppWithBackup(packageName: string, deviceId: string): Promise<void> {
  // 1. 导出 APK
  const apkPath = await invoke<string>('export_apk', { packageName, deviceId });
  
  // 2. 卸载应用
  await invoke('uninstall_app', { packageName, deviceId });
  
  // 3. 记录操作历史
  await operationHistoryService.recordOperation({
    type: 'uninstall',
    deviceId,
    parameters: { packageName },
    result: 'success',
    canUndo: true,
    undoData: { apkPath },
  });
}
```

**收益**:
- 降低误操作风险
- 提升用户信心
- 专业应用的标准功能

// src/hooks/useResponsiveLayout.ts（已存在，需增强）

export function useResponsiveLayout() {
  const [layout, setLayout] = useState<LayoutType>('desktop');
  const [scale, setScale] = useState<number>(1);

  useEffect(() => {
    const updateLayout = () => {
      const width = window.innerWidth;
      const dpr = window.devicePixelRatio;

      // 响应式断点
      if (width < 1024) {
        setLayout('compact');
      } else if (width < 1366) {
        setLayout('normal');
      } else {
        setLayout('desktop');
      }

      // 高DPI缩放
      if (dpr > 1.5) {
        setScale(Math.min(dpr / 1.5, 1.5));
      }
    };

    updateLayout();
    window.addEventListener('resize', updateLayout);
    return () => window.removeEventListener('resize', updateLayout);
  }, []);

  return { layout, scale };
}

## 5. 架构改进

### 5.1 前后端通信优化

**问题描述**: 频繁的 invoke 调用可能影响性能。

**技术方案**:

```typescript
// 实现请求队列和批处理
// src/services/core/tauriInvokeQueue.ts

class TauriInvokeQueue {
  private queue: Map<string, any[]> = new Map();
  private batchInterval = 50; // ms
  private timers: Map<string, NodeJS.Timeout> = new Map();

  async invoke<T>(command: string, args?: any): Promise<T> {
    // 对于可批处理的命令，加入队列
    if (this.isBatchable(command)) {
      return this.enqueueBatch(command, args);
    }

    // 其他命令直接调用
    return await tauriInvoke(command, args);
  }

  private isBatchable(command: string): boolean {
    return ['get_device_property', 'read_file_content'].includes(command);
  }

  private enqueueBatch<T>(command: string, args: any): Promise<T> {
    return new Promise((resolve, reject) => {
      if (!this.queue.has(command)) {
        this.queue.set(command, []);
      }

      this.queue.get(command)!.push({ args, resolve, reject });

      // 设置定时器批量执行
      if (this.timers.has(command)) {
        clearTimeout(this.timers.get(command)!);
      }

      const timer = setTimeout(() => {
        this.executeBatch(command);
      }, this.batchInterval);

      this.timers.set(command, timer);
    });
  }

  private async executeBatch(command: string) {
    const batch = this.queue.get(command);
    if (!batch || batch.length === 0) return;

    this.queue.delete(command);
    this.timers.delete(command);

    try {
      const batchArgs = batch.map(item => item.args);
      const results = await tauriInvoke(`batch_${command}`, { batch: batchArgs });

      batch.forEach((item, index) => {
        item.resolve(results[index]);
      });
    } catch (error) {
      batch.forEach(item => {
        item.reject(error);
      });
    }
  }
}

export const invokeQueue = new TauriInvokeQueue();

// 使用示例
const properties = await Promise.all([
  invokeQueue.invoke('get_device_property', { key: 'ro.product.model' }),
  invokeQueue.invoke('get_device_property', { key: 'ro.product.brand' }),
  invokeQueue.invoke('get_device_property', { key: 'ro.build.version.release' }),
]);
// 这些请求会被批处理为一次后端调用
```

### 5.2 缓存策略优化

**问题描述**: 缓存策略不够完善，缓存命中率低。

**技术方案**:

```typescript
// 实现多级缓存
// src/services/core/cacheService.ts

interface CacheItem<T> {
  data: T;
  timestamp: number;
  ttl: number;
  version: number;
}

class CacheService {
  private memoryCache: Map<string, CacheItem<any>> = new Map();
  private persistentCache: LocalForage;
  
  constructor() {
    this.persistentCache = localforage.createInstance({
      name: 'admt-cache',
    });
  }

  async get<T>(key: string, options?: CacheOptions): Promise<T | null> {
    // 1. 尝试从内存缓存获取
    const memCached = this.memoryCache.get(key);
    if (memCached && !this.isExpired(memCached)) {
      console.log(`Cache hit (memory): ${key}`);
      return memCached.data;
    }

    // 2. 尝试从持久化缓存获取
    if (options?.persistent !== false) {
      const diskCached = await this.persistentCache.getItem<CacheItem<T>>(key);
      if (diskCached && !this.isExpired(diskCached)) {
        console.log(`Cache hit (disk): ${key}`);
        // 回写到内存缓存
        this.memoryCache.set(key, diskCached);
        return diskCached.data;
      }
    }

    console.log(`Cache miss: ${key}`);
    return null;
  }

  async set<T>(key: string, data: T, options?: CacheSetOptions): Promise<void> {
    const item: CacheItem<T> = {
      data,
      timestamp: Date.now(),
      ttl: options?.ttl || 60000, // 默认1分钟
      version: options?.version || 1,
    };

    // 存入内存缓存
    this.memoryCache.set(key, item);

    // 存入持久化缓存
    if (options?.persistent !== false) {
      await this.persistentCache.setItem(key, item);
    }
  }

  async invalidate(key: string): Promise<void> {
    this.memoryCache.delete(key);
    await this.persistentCache.removeItem(key);
  }

  async invalidatePattern(pattern: string): Promise<void> {
    const regex = new RegExp(pattern);
    
    // 清除内存缓存
    for (const key of this.memoryCache.keys()) {
      if (regex.test(key)) {
        this.memoryCache.delete(key);
      }
    }

    // 清除持久化缓存
    const keys = await this.persistentCache.keys();
    for (const key of keys) {
      if (regex.test(key)) {
        await this.persistentCache.removeItem(key);
      }
    }
  }

  private isExpired(item: CacheItem<any>): boolean {
    return Date.now() - item.timestamp > item.ttl;
  }
}

export const cacheService = new CacheService();

// 使用示例：带缓存的设备信息获取
async function getDeviceInfoCached(deviceId: string): Promise<DeviceInfo> {
  const cacheKey = `device:${deviceId}:info`;
  
  // 尝试从缓存获取
  let info = await cacheService.get<DeviceInfo>(cacheKey);
  
  if (!info) {
    // 缓存未命中，从后端获取
    info = await invoke('get_device_info', { deviceId });
    
    // 存入缓存
    await cacheService.set(cacheKey, info, {
      ttl: 30000,  // 30秒
      persistent: true,
    });
  }
  
  return info;
}

// 设备断开连接时清除缓存
useEffect(() => {
  const unlisten = listen('device-disconnected', (event) => {
    const deviceId = event.payload.deviceId;
    cacheService.invalidatePattern(`device:${deviceId}:.*`);
  });

  return () => {
    unlisten.then(fn => fn());
  };
}, []);
```

**预估工作量**: 10-14小时

**收益**:
- 提高数据访问速度
- 减少后端调用
- 改善离线体验
